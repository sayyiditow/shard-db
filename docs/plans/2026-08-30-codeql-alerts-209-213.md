/* Append formatted text at *len without truncation. Returns 0 on success;

int tu_appendf(char *buf, size_t cap, size_t *len, const char *fmt, ...);

3. Add this complete implementation before `"int tu_run_cmd(const char *fmt, ...)"`:

```c
int tu_appendf(char *buf, size_t cap, size_t *len, const char *fmt, ...) {
    if (!buf || !len || !fmt || *len >= cap) return -1;

    va_list ap;
    va_start(ap, fmt);
    int n = vsnprintf(buf + *len, cap - *len, fmt, ap);
    va_end(ap);
    if (n < 0 || (size_t)n >= cap - *len) return -1;

    *len += (size_t)n;
    return 0;
}
```
